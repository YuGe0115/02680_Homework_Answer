\textbf{Problem 1. Tuples and Graphs}

\begin{enumerate}
    \item[(a)]
    \[
    \begin{aligned}
    E ={}&
    \left\{
    \{\langle p,q\rangle,\langle p,q+1\rangle\}
    \mid
    p\in\{1,\ldots,k\},\;
    q\in\{1,\ldots,k\}
    \right\} \\
    &\cup
    \left\{
    \{\langle p,q\rangle,\langle p+1,q\rangle\}
    \mid
    p\in\{1,\ldots,k-1\},\;
    q\in\{1,\ldots,k+1\}
    \right\}.
    \end{aligned}
    \]

    \item[(b)]
    Since
    \[
    V=\{1,\ldots,k\}\times\{1,\ldots,k+1\},
    \]
    the size of the Cartesian product is the product of the
    sizes of the two sets. Therefore,
    \[
    |V|
    =
    |\{1,\ldots,k\}|
    \cdot
    |\{1,\ldots,k+1\}|
    =
    k(k+1)
    =
    k^2+k.
    \]

    \item[(c)]
    For $k=2$,
    \[
    V=
    \{
    \langle1,1\rangle,
    \langle1,2\rangle,
    \langle1,3\rangle,
    \langle2,1\rangle,
    \langle2,2\rangle,
    \langle2,3\rangle
    \}.
    \]

\begin{center}
    \includegraphics[width=0.45\textwidth]{graph2.png}
\end{center}

    The degrees of the vertices are
    \[
    \operatorname{degree}_G(\langle1,1\rangle)=2,\qquad
    \operatorname{degree}_G(\langle2,1\rangle)=2,
    \]
    \[
    \operatorname{degree}_G(\langle1,2\rangle)=3,\qquad
    \operatorname{degree}_G(\langle2,2\rangle)=3,
    \]
    \[
    \operatorname{degree}_G(\langle1,3\rangle)=2,\qquad
    \operatorname{degree}_G(\langle2,3\rangle)=2.
    \]

    Thus,
    \[
    \sum_{v\in V}\operatorname{degree}_G(v)
    =
    4(2)+2(3)
    =
    14.
    \]

    The graph has four horizontal edges and three vertical edges,
    so
    \[
    |E|=4+3=7.
    \]
    Therefore,
    \[
    2|E|=2(7)=14
    =
    \sum_{v\in V}\operatorname{degree}_G(v).
    \]

\end{enumerate}

\textbf{Problem 2. Graphs and Matrices}

\begin{enumerate}
    \item[(a)]
    \[
    E=
    \{
    \langle S_1,L_1\rangle,
    \langle S_1,L_2\rangle,
    \langle S_2,L_1\rangle,
    \langle S_3,L_2\rangle
    \}.
    \]
    
    \[
    |N_G^{out}(S_1)|=2,\qquad
    |N_G^{out}(S_2)|=1,\qquad
    |N_G^{out}(S_3)|=1.
    \]
    
    \item[(b)]
    \[
    M=
    \begin{bmatrix}
    1 & 1\\
    1 & 0\\
    0 & 1
    \end{bmatrix}.
    \]
    
    \item[(c)]
    \[
    M\mathbf{1}
    =
    \begin{bmatrix}
    1 & 1\\
    1 & 0\\
    0 & 1
    \end{bmatrix}
    \begin{bmatrix}
    1\\
    1
    \end{bmatrix}
    =
    \begin{bmatrix}
    2\\
    1\\
    1
    \end{bmatrix}.
    \]
    
    The $i$th entry of $M\mathbf{1}$ equals
    $|N_G^{out}(S_i)|$ for $i=1,2,3$.
    Multiplying by $\mathbf{1}$ sums the entries in each row of $M$.
    Each $1$ in row $i$ represents an edge from $S_i$ to a lab,
    so the row sum equals $|N_G^{out}(S_i)|$.
\end{enumerate}

\textbf{Problem 3. Vectors and Matrices}

\begin{enumerate}
    \item[(a)]
    \[
    \|g_1\|_2=\sqrt{3^2+4^2+0^2}=5,
    \qquad
    \|g_2\|_2=\sqrt{0^2+3^2+4^2}=5.
    \]
    
    \[
    g_1\cdot g_2
    =3(0)+4(3)+0(4)
    =12.
    \]
    
    \item[(b)]
    \[
    M=
    \begin{bmatrix}
    3 & 4 & 0\\
    0 & 3 & 4
    \end{bmatrix},
    \qquad
    M^T=
    \begin{bmatrix}
    3 & 0\\
    4 & 3\\
    0 & 4
    \end{bmatrix}.
    \]
    
    \item[(c)]
    \[
    MM^T=
    \begin{bmatrix}
    3 & 4 & 0\\
    0 & 3 & 4
    \end{bmatrix}
    \begin{bmatrix}
    3 & 0\\
    4 & 3\\
    0 & 4
    \end{bmatrix}
    =
    \begin{bmatrix}
    25 & 12\\
    12 & 25
    \end{bmatrix}.
    \]
    
    The diagonal entries are
    $\|g_1\|_2^2=\|g_2\|_2^2=25$,
    and the off-diagonal entries are
    $g_1\cdot g_2=g_2\cdot g_1=12$.
    
    Also,
    \[
    (MM^T)^T=MM^T,
    \]
    so $MM^T$ is symmetric. This makes sense since
    \[
    (MM^T)_{ij}=g_i\cdot g_j
    \]
    and $g_i\cdot g_j=g_j\cdot g_i$.
\end{enumerate}

\textbf{Problem 4. Matrices and Linear Systems}

\begin{enumerate}
    \item[(a)]
    \[
    A=
    \begin{bmatrix}
    1 & 2\\
    3 & -1
    \end{bmatrix},
    \qquad
    \mathbf{x}=
    \begin{bmatrix}
    x\\
    y
    \end{bmatrix},
    \qquad
    b=
    \begin{bmatrix}
    5\\
    1
    \end{bmatrix}.
    \]

    \[
    A\mathbf{x}=b.
    \]
    
    \item[(b)]
    \[
    |A|=1(-1)-2(3)=-7.
    \]
    
    To compute $A^{-1}$,
    \[
    \left[
    \begin{array}{cc|cc}
    1 & 2 & 1 & 0\\
    3 & -1 & 0 & 1
    \end{array}
    \right]
    \xrightarrow{R_2\leftarrow R_2-3R_1}
    \left[
    \begin{array}{cc|cc}
    1 & 2 & 1 & 0\\
    0 & -7 & -3 & 1
    \end{array}
    \right]
    \]
    
    \[
    \xrightarrow{R_2\leftarrow-\frac{1}{7}R_2}
    \left[
    \begin{array}{cc|cc}
    1 & 2 & 1 & 0\\
    0 & 1 & \frac{3}{7} & -\frac{1}{7}
    \end{array}
    \right]
    \xrightarrow{R_1\leftarrow R_1-2R_2}
    \left[
    \begin{array}{cc|cc}
    1 & 0 & \frac{1}{7} & \frac{2}{7}\\
    0 & 1 & \frac{3}{7} & -\frac{1}{7}
    \end{array}
    \right].
    \]
    
    Therefore,
    \[
    A^{-1}
    =
    \begin{bmatrix}
    \frac{1}{7} & \frac{2}{7}\\
    \frac{3}{7} & -\frac{1}{7}
    \end{bmatrix}.
    \]
    
    \item[(c)]
    \[
    \mathbf{x}
    =
    A^{-1}b
    =
    \begin{bmatrix}
    \frac{1}{7} & \frac{2}{7}\\
    \frac{3}{7} & -\frac{1}{7}
    \end{bmatrix}
    \begin{bmatrix}
    5\\
    1
    \end{bmatrix}
    =
    \begin{bmatrix}
    1\\
    2
    \end{bmatrix}.
    \]
    
    Therefore,
    \[
    x=1,\qquad y=2.
    \]
    
    Verification:
    \[
    1+2(2)=5,
    \qquad
    3(1)-2=1.
    \]
\end{enumerate}
